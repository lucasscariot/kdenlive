cmake_minimum_required(VERSION 3.22)
project(KdenliveMcpProtocolTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS HttpServer Network Test)
enable_testing()
add_executable(mcp_protocol_test protocoltest.cpp ../../src/mcp/mcpserver.cpp)
target_include_directories(mcp_protocol_test PRIVATE ../../src/mcp)
target_link_libraries(mcp_protocol_test PRIVATE Qt6::HttpServer Qt6::Network Qt6::Test)
target_compile_definitions(mcp_protocol_test PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_NO_KEYWORDS)
add_test(NAME mcp_protocol COMMAND mcp_protocol_test)
